Private Sub ListBox1_KeyDown(ByVal KeyCode As MSForms.ReturnInteger, ByVal Shift As Integer)
  If ListBox1.ListIndex = -1 And ListBox1.ListCount > 0 Then ListBox1.ListIndex = 0
  If KeyCode = 27 Then Me.Hide
  If KeyCode = 13 Then
    If ListBox1.ListIndex = -1 Then ListBox1.ListIndex = 0
    If CheckBox1.Value = True Then
      TextBox3.Text = Left(ListBox1.List(ListBox1.ListIndex, 2), 8)
      TextBox2.Text = Mid(ListBox1.List(ListBox1.ListIndex, 2), 10)
      CheckBox1.Value = False
      TextBox2.SetFocus
    Else
      SendKeys "{TAB}"
    End If
  End If

Private Sub WebBrowser2_DocumentComplete(ByVal pDisp As Object, URL As Variant)
  Dim f As String
  
  On Error Resume Next
  f = WebBrowser2.Document.Body.InnerText
  oc1 = 0
  oc2 = 0
  oc3 = 0
  'TextBox4.Text = ""
  For n = 1 To Len(f)
    
    If Mid(f, n, 9) = "<NmbItem>" Then z1 = n + 9
    If Mid(f, n, 10) = "</NmbItem>" Then z2 = n
    If z2 > 0 Then
      b = b & Mid(f, z1, z2 - z1) & "; "
      z2 = 0
    End If
    If Mid(f, n, 9) = "<DscItem>" Then z1 = n + 9
    If Mid(f, n, 10) = "</DscItem>" Then z2 = n
    If z2 > 0 Then
      b = Trim(b) & Mid(f, z1, z2 - z1) & "; "
      z2 = 0
    End If
    
    If Mid(f, n, 14) = "<TpoDocRef>801" Then oc1 = 1
    If Mid(f, n, 10) = "<FolioRef>" And oc1 > 0 Then oc2 = n + 10
    If Mid(f, n, 11) = "</FolioRef>" And oc2 > 0 Then
      'MsgBox oc1 & " - " & oc2 & " - " & n
      'MsgBox Mid(f, oc2, n - oc2)
      TextBox4.Text = Mid(f, oc2, n - oc2)
      oc1 = 0
      oc2 = 0
    End If
    
    If Mid(f, n, 10) = "<FmaPago>1" Then
      b = "CAJA - " & b
      MsgBox "Pago Contado" & Space(10), vbInformation, "JCV"
    End If
    
  Next
  
  TextBox2.Text = Trim(Left(b, Len(b) - 1))
  Label15.Caption = ""
  Exit Sub

End Sub
